cmake_minimum_required(VERSION 3.18)
project(TangoPinyinTIP LANGUAGES CXX)

add_library(TangoPinyinTIP SHARED tip.cpp)
target_compile_features(TangoPinyinTIP PRIVATE cxx_std_17)
target_link_libraries(TangoPinyinTIP PRIVATE ole32 uuid advapi32 user32)
set_target_properties(TangoPinyinTIP PROPERTIES PREFIX "")
if(MSVC)
    set_property(TARGET TangoPinyinTIP PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(TangoPinyinTIP PRIVATE /utf-8)
elseif(MINGW)
    target_link_options(TangoPinyinTIP PRIVATE -static -static-libgcc -static-libstdc++)
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
        target_link_options(TangoPinyinTIP PRIVATE -Wl,--kill-at)
    endif()
endif()
